## Task B — Project grid: top-8 default, "Show all (N) ▾" expand

**File:** `web/src/components/Home/index.tsx`, `Home.css`

### Step B.1 — Sort + slice in the parent

**Sort key — relevance, not total volume.** The grid surfaces *what's likely to need attention now*, not *what was historically busy*:

1. Has live (running / active) sessions → first
2. Has waiting / disconnected sessions → next
3. Most recent activity (`lastEventAt`) → next
4. Total session count → tiebreaker only

In `Home/index.tsx`:

```tsx
const [projectsExpanded, setProjectsExpanded] = useState(false);
const PROJECTS_DEFAULT_VISIBLE = 8;

// Recency per project (max lastEventAt across its sessions).
const lastActivityByProject = useMemo(() => {
  const out: Record<string, number> = {};
  for (const s of sessions) {
    if (!s.projectId) continue;
    const t = parseTimestamp(s.lastEventAt);
    if (!Number.isFinite(t)) continue;
    if (t > (out[s.projectId] ?? 0)) out[s.projectId] = t;
  }
  return out;
}, [sessions]);

// Total per project (live + done) — only used as the final tiebreaker.
const sessionTotalsByProject = useMemo(() => {
  const out: Record<string, number> = {};
  for (const [pid, c] of Object.entries(sessionCountsByProject)) {
    out[pid] = (c.running ?? 0) + (c.active ?? 0) + (c.waiting ?? 0) + (c.disconnected ?? 0) + (c.done ?? 0);
  }
  return out;
}, [sessionCountsByProject]);

const sortedProjects = useMemo(() => {
  const score = (id: string) => {
    const c = sessionCountsByProject[id] ?? {};
    return {
      hasLive: ((c.running ?? 0) + (c.active ?? 0)) > 0 ? 1 : 0,
      hasWaiting: ((c.waiting ?? 0) + (c.disconnected ?? 0)) > 0 ? 1 : 0,
      lastActivity: lastActivityByProject[id] ?? 0,
      total: sessionTotalsByProject[id] ?? 0,
    };
  };
  return [...projects].sort((a, b) => {
    const sa = score(a.id);
    const sb = score(b.id);
    if (sa.hasLive !== sb.hasLive) return sb.hasLive - sa.hasLive;
    if (sa.hasWaiting !== sb.hasWaiting) return sb.hasWaiting - sa.hasWaiting;
    if (sa.lastActivity !== sb.lastActivity) return sb.lastActivity - sa.lastActivity;
    return sb.total - sa.total;
  });
}, [projects, sessionCountsByProject, lastActivityByProject, sessionTotalsByProject]);

const visibleProjects = projectsExpanded
  ? sortedProjects
  : sortedProjects.slice(0, PROJECTS_DEFAULT_VISIBLE);

const hiddenCount = sortedProjects.length - visibleProjects.length;
```

`parseTimestamp` already exists in the file — find via grep, reuse. If not, `Date.parse(s.lastEventAt)` is fine.

## Task C — Sessions: extract `SessionRow`, add FULL view + toggle

### Step C.1 — Extract `SessionRow`

Today the sessions table renders rows inline in `Home/index.tsx`. Extract into a component that accepts a `view: "full" | "compact"` prop.

```tsx
// web/src/components/Home/SessionRow.tsx
import type { Session } from "../../data/sessions-api";
import type { Artifact } from "../../data/artifacts-api";
import { humanRelativeTime } from "../../lib/time";  // existing util — confirm name

interface SessionRowProps {
  session: Session;
  view: "full" | "compact";
  artifacts: Artifact[];  // artifacts attributed to this session (empty array if none)
  onOpen: (sessionId: string) => void;
}

export function SessionRow({ session, view, artifacts, onOpen }: SessionRowProps) {
  if (view === "compact") return <CompactRow session={session} onOpen={onOpen} />;
  return <FullRow session={session} artifacts={artifacts} onOpen={onOpen} />;
}
```

Implement `CompactRow` (the existing single-line layout — port from current inline JSX) and `FullRow` (the two-line layout from `home-full-page.html` mock).

const artifactsBySessionId = useMemo(() => {
